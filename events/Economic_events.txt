add_namespace = Economic_events

#Reset Year inflation
country_event = {
	id = Economic_events.1
	title = Economic_events.1.t
	desc = Economic_events.1.d
	#picture = GFX_report_event_diplomatic_message

	is_triggered_only = yes
	
	fire_only_once = yes
	
	hidden = yes
	
	trigger = {
	}

	option = {
		name = Economic_events.1.a
		UKR = {
			set_variable = {
				var = year_inflation
				value = 0
			}
		}
		RUS = {
			set_variable = {
				var = year_inflation
				value = 0
			}
		}
		set_variable = { cash = 0 }
		if = {
			limit = {
				check_variable = {
					var = cash
					value = 0
					compare = less_than
				}
			}
			set_temp_variable = { temp_debt = var:cash }
			multiply_variable = { var = temp_debt value = -1 }
			add_to_variable = { var = debt value = var:temp_debt }
			set_variable = { var = cash value = 1 }
			clear_variable = temp_debt
		}
		country_event = {
			id = Economic_events.1
			days = 365
		}
		
	}
}
