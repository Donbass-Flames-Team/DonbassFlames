on_actions = {

	on_daily = {
		effect = {
			# 1. Выборы 2019 года
			if = {
				limit = {
					tag = UKR
					date > 2018.12.30
					date < 2019.1.1
				}
				country_event = { id = ukr_elections_2019.1 }
			}
			
			# 2. События Шатерска (2014)
			if = {
				limit = {
					tag = UKR
					date > 2014.10.15
					NOT = {
						has_country_flag = Shatersk_disbanded
					}
				}
				set_country_flag = Shatersk_disbanded
				country_event = { id = ukraine.137 }
			}
			
			# 3. Смерть Андрея Кузьменко (Скрябин) — 2 февраля 2015 года
			if = {
				limit = {
					tag = UKR
					date > 2015.2.1
					date < 2015.2.3
				}
				country_event = { id = ukraine.2015 }
			}
		}
	}
}
