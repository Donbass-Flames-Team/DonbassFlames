	on_startup = {
		UKR = {
		country_event = { id = ukr_elections_2019.1 days = 1866 } 
		country_event = { id = ukraine.160 days = 438 }
		country_event = { id = ukraine.172 days = 972 }
		country_event = { id = ukraine.173 days = 1096 }
            }
}
	on_daily = {
		effect = {
			if = {
				limit = {
					tag = UKR
					date > 2014.10.15
					NOT = { has_country_flag = Shatersk_disbanded }
				}
				set_country_flag = Shatersk_disbanded
				country_event = { id = ukraine.137 }
			}
			
            # 17.10.2017 — ПРОТЕСТЫ У ВЕРХОВНОЙ РАДЫ
            if = {
                limit = {
					tag = UKR
                    date > 2017.10.16
                    date < 2017.10.18

                    NOT = {
                        has_country_flag = UKR_rada_protests
                    }
                }
                set_country_flag = UKR_rada_protests

                # СЛАБЫЕ ПРОТЕСТЫ
                # Стабильность > 80%

                if = {
                    limit = {
                        has_stability > 0.80
                    }
                    country_event = {
                        id = ukraine.174
                    }
                }


                # СРЕДНИЕ ПРОТЕСТЫ
                # Стабильность 40–80%

                else_if = {
                    limit = {
                        has_stability > 0.40
                    }
                    country_event = {
                        id = ukraine.175
                    }
                }


                # СИЛЬНЫЕ ПРОТЕСТЫ
                # Стабильность < 40%

                else = {
                    country_event = {

                    }
                }
            }
		}
	}
