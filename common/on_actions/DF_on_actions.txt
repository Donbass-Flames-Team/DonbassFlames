on_actions = {
	on_startup = {
		effect = {
			every_country = {
				country_event = {
					id = Economic_events.2
					days = 1
				}
			}
		}
	}
	on_daily = {
		effect = {
			###############################
			#### CUSTOM BUILDING UTILS ####
			###############################
			DF_buildings_setup = yes
		}
	}
	on_monthly = {
		effect = {
			###################
			### THE ECONOMY ###
			###################
			set_variable = {
		        var = gdp_last_month
		        value = gdp
	        }
			DF_economy_setup = yes
			set_global_flag = add_remove_cash_flag
			set_variable = {
				var = gdp_current_month
				value = gdp
			}			
			subtract_from_variable = {
				var = gdp_current_month
				value = gdp_last_month
			}
			divide_variable = {
				var = gdp_current_month
				value = gdp_last_month
			}
			multiply_variable = {
				var = gdp_current_month
				value = 100				
			}
		}
	}
	on_weekly = {
		effect = {
	        DF_key_rate_calc = yes
		}
	}
	on_weekly = {
		effect = {
			if = {
				limit = {
					check_variable = {
						var = cash
						value = 0
						compare = less_than
					}
				}
				set_temp_variable = { temp_debt = var:cash }
				multiply_variable = { var = temp_debt value = -1 }
				add_to_variable = { var = debt value = var:temp_debt }
				set_variable = { var = cash value = 1 }
				clear_variable = temp_debt
			}	
		}	
	}
	on_government_change = {
		effect = {
			if = {
				limit = {
					original_tag = UKR
				}
				if = {
					limit = {
						has_country_leader = {
							character = UKR_victor_yanukovich
							ruling_only = yes
						}
						RUS = {
							has_country_leader = {
								character = vladimir_putin
								ruling_only = yes
							}
						}
					}
					add_opinion_modifier = {
						target = RUS
						modifier = kharkiv_pact
					}
					reverse_add_opinion_modifier = {
						target = RUS
						modifier = kharkov_accords
					}
					RUS = {
						diplomatic_relation = {
							country = UKR
							relation = docking_rights
						}
					}
					else_if = {
						limit = {
							RUS = {
								has_completed_focus = RUS_terminate_kharkov_accords
							}
						}
						remove_opinion_modifier = {
							target = RUS
							modifier = kharkiv_pact
						}
						RUS = {
							remove_opinion_modifier = {
								target = UKR
								modifier = kharkov_accords
							}
							diplomatic_relation = {
								country = UKR
								relation = docking_rights
								active = no
							}
						}
					}
				}
			}
		}
	}

}
