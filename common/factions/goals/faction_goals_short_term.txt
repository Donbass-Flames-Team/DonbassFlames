faction_goal_unlock_research = {
	name = faction_goal_unlock_research_name
	description = faction_goal_unlock_research_desc
	category = short_term
	group = FOCUS_FILTER_INTERNAL_AFFAIRS
	available = {
		FROM = { num_faction_members > 2 }
	}
	visible = {
		FROM = { has_faction_research_unlocked = no }
	}
	select_effect = {
		scope:faction = {
			set_temp_variable = {
				total_techs = {
					value = 0
					every_collection = {
						named_collection = faction_members
						add = {
							value = num_researched_technologies
							multiply = {
								value = 0
								if = { limit = { value = num_researched_technologies  less_than = 50 }  add = 1.25 }
								else_if = { limit = { value = num_researched_technologies  less_than = 75 }  add = 1.2 }
								else_if = { limit = { value = num_researched_technologies  less_than = 100 }  add = 1.15 }
								else_if = { limit = { value = num_researched_technologies  less_than = 150 }  add = 1.1 }
								else_if = { limit = { value = num_researched_technologies  less_than = 200 }  add = 1.05 }
							}
						}
					}
					round = yes
				}
			}
			set_variable = { unlock_military_techs_researched_target = total_techs }
		}
	}
	completed = {
		scope:faction = {
			hidden_trigger = {
				tooltip_evaluation = eval
				set_temp_variable = {
					unlock_military_techs_researched_compare_temp_target = {
						value = 0
						every_collection = {
							named_collection = faction_members
							add = num_researched_technologies
						}
					}
				}
				set_temp_variable = {
					unlock_military_techs_researched_compare = {
						value = 0
						every_collection = {
							named_collection = faction_members
							add = {
								value = num_researched_technologies
								multiply = {
									value = 0
									if = { limit = { value = num_researched_technologies  less_than = 50 }  add = 1.25 }
									else_if = { limit = { value = num_researched_technologies  less_than = 75 }  add = 1.2 }
									else_if = { limit = { value = num_researched_technologies  less_than = 100 }  add = 1.15 }
									else_if = { limit = { value = num_researched_technologies  less_than = 150 }  add = 1.1 }
									else_if = { limit = { value = num_researched_technologies  less_than = 200 }  add = 1.05 }
								}
							}
						}
						round = yes
					}
				}
			}

			if = {
				limit = {
					has_variable = unlock_military_techs_researched_target
				}
				check_variable = {
					var = unlock_military_techs_researched_compare_temp_target
					value = unlock_military_techs_researched_target
					compare = greater_than
					tooltip = "GOAL_T_TRIGGER_A"
				}
			}
			else = {
				check_variable = {
					value = unlock_military_techs_researched_compare
					var = unlock_military_techs_researched_compare_temp_target
					compare = greater_than
					tooltip = "GOAL_T_TRIGGER_A"
				}
			}
		}
	}
	complete_effect = {
		set_faction_research_unlocked = yes
		add_faction_initiative = 1
	}
}

faction_goal_spy_master = {
	name = spy_master_goal_name
	description = spy_master_goal_desc
	group = FOCUS_FILTER_POLITICAL_CHARACTER
	category = short_term
	allowed = {
		has_dlc = "La Resistance"
	}
	visible = {
		FROM = {
			is_spymaster = no
			any_allied_country = {
				is_spymaster = no
			}
		}
	}
	completed = {
		OR = {
			is_spymaster = yes
			any_allied_country = {
				is_spymaster = yes
			}
		}
	}
	complete_effect = {
		add_faction_power_projection = 50
		custom_effect_tooltip = generic_skip_one_line_tt
		custom_effect_tooltip = THE_SPY_MASTER
		add_faction_initiative = 1
		custom_effect_tooltip = generic_skip_one_line_tt
		every_faction_member = {
			limit = {
				has_intelligence_agency = no
			}
			IF = {
				limit = { NOT = { has_completed_goal = faction_goal_spy_master } }
				create_intelligence_agency = yes
			}
		}
		every_faction_member = {
			limit = {
				has_intelligence_agency = yes
				NOT = { has_done_agency_upgrade  = upgrade_form_department }
			}
			IF = {
				limit = { NOT = { has_completed_goal = faction_goal_spy_master } }
				upgrade_intelligence_agency = upgrade_form_department
			}
		}
		every_faction_member = {
			limit = {
				has_intelligence_agency = yes
				has_done_agency_upgrade  = upgrade_form_department
				NOT = { has_done_agency_upgrade = upgrade_crypto_strength }
			}
			IF = {
				limit = { NOT = { has_completed_goal = faction_goal_spy_master } }
				upgrade_intelligence_agency = upgrade_crypto_strength
			}
		}
		every_faction_member = {
			limit = {
				has_intelligence_agency = yes
				has_done_agency_upgrade = upgrade_crypto_strength
				has_tech = mechanical_computing
				NOT = { has_done_agency_upgrade = upgrade_crypto_strength_2 }
			}
			add_faction_initiative = 1
			IF = {
				limit = { NOT = { has_completed_goal = faction_goal_spy_master } }
				upgrade_intelligence_agency = upgrade_crypto_strength_2
			}
		}
		every_faction_member = {
			limit = {
				has_intelligence_agency = yes
				NOT = { agency_upgrade_number > 30 }
				OR = {
					has_done_agency_upgrade = upgrade_crypto_strength_2
					NOT = { has_tech = mechanical_computing }
				}
			}
			add_faction_initiative = 1
			IF = {
				limit = { NOT = { has_completed_goal = faction_goal_spy_master } }
				gain_random_agency_upgrade = yes
			}
		}
		every_faction_member = { # Fallback in case someone completed everything
			visible_when_empty = no
			limit = {
				has_intelligence_agency = yes
				agency_upgrade_number > 30
			}
			add_faction_initiative = 1
			IF = {
				limit = { NOT = { has_completed_goal = faction_goal_spy_master } }
				add_political_power = 100
			}
		}
	}
	ai_will_do = {
		base = 1
		modifier = {
			add = 2
			has_intelligence_agency = yes
		}
	}
}
