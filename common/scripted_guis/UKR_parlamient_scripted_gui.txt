scripted_gui = {

	UKR_parlamient_ui_window = {
		context_type = decision_category
		window_name = "UKR_parlamient_ui_window"

		effects = {
			UKR_rada_button_click = {
				if = {
					limit = { has_variable = show_parlamient_general_window }
					clear_variable = show_parlamient_general_window
				}
				else = {
					set_variable = { show_parlamient_general_window = 1 }
				}
			}
		}

		properties = {
			senate_icon = {
				frame = current_senate_icon_frame
			}

			house_icon = {
				frame = current_house_icon_frame
			}
		}
	}

	UKR_parlamient_general_window = {
		context_type = player_context
		parent_window_token = top_bar
		window_name = "UKR_parlamient_general_window"
		dirty = show_parlamient_general_window

		visible = {
			check_variable = { show_parlamient_general_window > 0 }
		}
		
		effects = {
			close_map_button_click = {
				clear_variable = show_parlamient_general_window
			}
		}
		
		dynamic_lists = {
			parliament_main_grid = {
				array = parliament_main_array
				index = parliament_main_index
				change_scope = no
				entry_container = parliament_main_container
			}
			parliament_semicircle_main_grid = {
				array = parliament_main_array
				index = parliament_main_index
				change_scope = no
				entry_container = parliament_semicircle_main_entry
			}
			parliament_member_grid = {
				array = parliament_member_array
				index = parliament_member_index
				change_scope = no
				entry_container = parliament_member_entry
			}
		}
		
		properties = {
			UKR_speaker_parlamient_portrait = {
				image = "[ukr_parlamient_speaker_image]"
			}
			parliament_main_icon = {
				frame = v
			}
			parliament_semicircle_main_icon = {
				frame = v
				x = parliament_x_array^parliament_main_index
				y = parliament_y_array^parliament_main_index
			}
			parliament_member_icon = {
				frame = v
			}
		}
	}
	
}
