defined_text = {
	name = economy_status


	text = {
		trigger = {
			check_variable = {
				var = gdp_current_month
				value = -5
				compare = less_than_or_equals
			}
		}
		localization_key = "economy_status_depression"
	}

	text = {
		trigger = {
			check_variable = {
				var = gdp_current_month
				value = -5
				compare = greater_than
			}
			check_variable = {
				var = gdp_current_month
				value = -1
				compare = less_than
			}
		}
		localization_key = "economy_status_recession"
	}

	text = {
		trigger = {
			check_variable = {
				var = gdp_current_month
				value = -1
				compare = greater_than_or_equals
			}
			check_variable = {
				var = gdp_current_month
				value = 1
				compare = less_than_or_equals
			}
		}
		localization_key = "economy_status_stagnation"
	}

	text = {
		trigger = {
			check_variable = {
				var = gdp_current_month
				value = 1
				compare = greater_than
			}
			check_variable = {
				var = gdp_current_month
				value = 4
				compare = less_than_or_equals
			}			
		}
		localization_key = "economy_status_steady_growth"
	}

	text = {
		trigger = {
			check_variable = {
				var = gdp_current_month
				value = 4
				compare = greater_than
			}
			check_variable = {
				var = gdp_current_month
				value = 7
				compare = less_than_or_equals
			}			
		}
		localization_key = "economy_status_fast_growth"
	} 
	
	text = {
		trigger = {
			check_variable = {
				var = gdp_current_month
				value = 7
				compare = greater_than
			}			
		}
		localization_key = "economy_status_economic_boom"
	}	

}
